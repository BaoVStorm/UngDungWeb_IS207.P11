--------- Selector
* 		// all
tag
t1 t2		// chọn tất cả t2 trong t1
t1, t2		// chọn tất cả t1 và t2
t1 + t2		// chọn tất cả t2 kế sau t1
t1 > t2		// chọn tất cả t2 có cha là t1
.class
#id

[attribute]
div[class$="hinh"] 	// Chọn các thành phần với thuộc tính kết thúc bằng “giá trị”
[title~="myWeb"] 		// Chọn các thành phần với thuộc tính chứa một “giá trị”
[target=_"blank"]		// Chọn các thành phần với thuộc tính bằng “giá trị”.
[title^="myWeb"]		// Chọn các thành phần với thuộc tính bắt đầu bằng “giá trị”.
[class*="te"]		// Chọn các thành phần với thuộc tính có chứa “giá trị” ở bất kỳ đâu trong thuộc tính.

	-- Pseudo-classes --

:after			// Thêm nội dung ngay phía sau thành phần.
:before 		// Thêm nội dung ngay phía trước thành phần.
:first-child		// Chọn thành phần đầu tiên của thành phần cha chứa nó.
:first-letter 		// Chọn ký tự đầu tiên của một thành phần.
:first-line 		// Chọn dòng đầu tiên của một thành phần.
:focus 			// Thành phần sẽ focus khi được chọn.

:link 			// Định dạng cho tất cả liên kết khi chưa được click.
:visited 		// Định dạng cho các thành phần đã được click.
:hover 			// Chọn các liên kết được hover (di chuyển chuột lên thành phần).
:active			// Chọn các liên kết được kích hoạt.
